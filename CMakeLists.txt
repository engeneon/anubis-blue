cmake_minimum_required(VERSION 3.24)
project(anubis VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(anubis_core src/domain/quantity.cpp src/domain/json.cpp src/compiler/compiler.cpp src/compiler/planning.cpp src/compiler/qualification.cpp src/compiler/resolution.cpp src/compiler/resources.cpp src/application/knowledge/resolve.cpp src/application/intake/import.cpp src/adapters/projection.cpp)
target_include_directories(anubis_core PUBLIC src third_party)
target_link_libraries(anubis_core PUBLIC OpenSSL::Crypto)
target_compile_options(anubis_core PRIVATE -Wall -Wextra -Wconversion)
add_executable(anubis src/api/main.cpp src/adapters/neon_transport.cpp src/application/service.cpp src/application/copilot/service.cpp src/adapters/model_transport.cpp src/persistence/store.cpp src/persistence/postgresql/backend.cpp)
target_link_libraries(anubis PRIVATE anubis_core SQLite::SQLite3 Threads::Threads)
target_compile_definitions(anubis PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(anubis PRIVATE OpenSSL::SSL)
add_executable(anubis_tests tests/core.cpp)
target_link_libraries(anubis_tests PRIVATE anubis_core)
enable_testing()
add_test(NAME core COMMAND anubis_tests ${CMAKE_SOURCE_DIR})
install(TARGETS anubis RUNTIME DESTINATION bin)
install(DIRECTORY ui/dist/ DESTINATION share/anubis/ui/dist)
install(DIRECTORY contracts knowledge DESTINATION share/anubis)
install(DIRECTORY tests/fixtures/ DESTINATION share/anubis/tests/fixtures)
file(GLOB_RECURSE ANUBIS_ENGINE_INPUTS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/compiler/*" "${CMAKE_SOURCE_DIR}/src/domain/*" "${CMAKE_SOURCE_DIR}/src/adapters/projection.cpp")
list(SORT ANUBIS_ENGINE_INPUTS)
set(ANUBIS_ENGINE_HASH_INPUT "")
foreach(INPUT_FILE IN LISTS ANUBIS_ENGINE_INPUTS)
  file(SHA256 "${INPUT_FILE}" INPUT_HASH)
  string(APPEND ANUBIS_ENGINE_HASH_INPUT "${INPUT_HASH}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ANUBIS_ENGINE_INPUTS})
string(SHA256 ANUBIS_ENGINE_SHA256 "${ANUBIS_ENGINE_HASH_INPUT}")
target_compile_definitions(anubis_core PUBLIC ANUBIS_ENGINE_SHA256="${ANUBIS_ENGINE_SHA256}")
option(ANUBIS_SANITIZE "Address and undefined behavior sanitizers" OFF)
if(ANUBIS_SANITIZE)
  foreach(TARGET anubis_core anubis anubis_tests)
    target_compile_options(${TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()

option(ANUBIS_WITH_POSTGRES "Enable experimental PostgreSQL repository" OFF)
if(ANUBIS_WITH_POSTGRES)
  find_package(PostgreSQL REQUIRED)
  target_compile_definitions(anubis PRIVATE ANUBIS_WITH_POSTGRES)
  target_link_libraries(anubis PRIVATE PostgreSQL::PostgreSQL)
endif()

add_executable(anubis_characterization_tests tests/characterization.cpp)
target_link_libraries(anubis_characterization_tests PRIVATE anubis_core)
add_test(NAME characterization COMMAND anubis_characterization_tests ${CMAKE_SOURCE_DIR})
add_executable(anubis_guided_driver tests/guided_driver.cpp)
target_link_libraries(anubis_guided_driver PRIVATE anubis_core)
